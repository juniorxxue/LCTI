\subsection{Typing: $\Gamma \vdash_n e : A$}

\begin{mathpar}
    \inferrule*[right=\texttt{Int}]
	{ }
    {\Gamma \vdash_0 i : \mathtt{Int}}

    \inferrule*[right=\texttt{Var}]
    { }
    {\Gamma \vdash_0 x : A}

    \inferrule*[right=\texttt{Ann}]
    {\Gamma \vdash_{\infty} e : A}
    {\Gamma \vdash_0 (e : A) : A}

    \inferrule*[right=\texttt{Lam}]
    {\Gamma, x : A \vdash_{n-} e : B}
    {\Gamma \vdash_{n} \lambda x.~ e : A \rightarrow B}
    \\

    \inferrule*[right=\texttt{App1}]
    {\Gamma \vdash_{C~n} e_1 : A \rightarrow B \\
    \Gamma \vdash_{\infty} e_2 : A}
    {\Gamma \vdash_n e_1~e_2 : B}

    \inferrule*[right=\texttt{App2}]
    {\Gamma \vdash_{I~n} e_1 : A \rightarrow B \\
    \Gamma \vdash_0 e_2 : A}
    {\Gamma \vdash_n e_1~e_2 : B}
    \\

    \inferrule*[right=\texttt{Sub}]
    {\Gamma \vdash_0 e : B \\
    n \neq 0\\
    \Gamma \vdash_n B <: A}
    {\Gamma \vdash_n e : A}

    \inferrule*[right=\texttt{TAbs}]
    {\Gamma, a \vdash_0 e : A}
    {\Gamma \vdash_0 \Lambda a.~e : \forall a.~A}
\end{mathpar}

\section{Algo.}

\begin{align*}
&\text{Typing Environments} \quad\quad &\Gamma::=&~ . \mid \Gamma, x : A \mid \Gamma, a \mid \Gamma, \hat{a} = A\\
&\text{Subtyping Environments} \quad\quad &\Gamma, \Omega, \Delta ::=&~ . \mid \Gamma, x : A \mid \Gamma, a \mid \Gamma , \hat{a} \mid \Gamma, \hat{a} = A\\
    &\text{Contexts} \quad\quad &\Sigma ::=&~  \square  \mid A \mid \boxed{e} \leadsto \Sigma\\
    &\text{Generic Consumers} \quad \quad &g::=&~ i \mid x \mid e : A \mid \Lambda a.~e
\end{align*}

\subsection{Subtyping: $\boxed{\Gamma \vdash A <: \Sigma \dashv \Gamma' \rightsquigarrow A}$}

\begin{mathpar}
\inferrule*[right=\texttt{S-Int}]
{ }
{\Gamma \vdash \mathtt{Int} <: \mathtt{Int} \dashv \Gamma \rightsquigarrow \mathtt{Int}}

\inferrule*[right=\texttt{S-Empty}]
{FV_{ex}(\Gamma, A) = . \\ \Gamma \vdash A}
{\Gamma \vdash A <: \square \dashv \Gamma \rightsquigarrow A}

\inferrule*[right=\texttt{S-Ex-L}]
{FV_{ex}(\Gamma, A) = . \\ \Gamma \vdash A}
{\Gamma \vdash \hat{a} <: A \dashv [A/\hat{a}]~\Gamma \rightsquigarrow A}

\inferrule*[right=\texttt{S-Ex-R}]
{FV_{ex}(\Gamma, A) = . \\ \Gamma \vdash A}
{\Gamma \vdash A <: \hat{a} \dashv [A/\hat{a}]~\Gamma \rightsquigarrow A}

\inferrule*[right=\texttt{S-Arr}]
{\Gamma \vdash C <: A \dashv \Gamma_1 \rightsquigarrow A' \\
 \Gamma_1 \vdash B <: D \dashv \Gamma_2 \rightsquigarrow D'}
{\Gamma \vdash A \rightarrow B <: C \rightarrow D \dashv \Gamma_2 \rightsquigarrow C \rightarrow D}

\inferrule*[right=\texttt{S-Hole-No-Ex}]
{ FV_{ex}(\Gamma, A) = . \\
\Gamma \vdash A \Rightarrow e \Rightarrow C\\
 \Gamma \vdash B <: \Sigma \dashv \Gamma' \rightsquigarrow D}
{\Gamma \vdash A \rightarrow B <: \boxed{e} \leadsto \Sigma \dashv \Gamma' \rightsquigarrow A \rightarrow D}

\inferrule*[right=\texttt{S-Hole-Ex}]

\Gamma_1 \vdash B <: \Sigma \dashv \Gamma_2 \rightsquigarrow D}

\subsection{Subtyping: $\Gamma; \Delta \vdash A \leq^{\pm} \Sigma \dashv \Gamma; \Delta' \rightsquigarrow A'$}

\begin{mathpar}
\inferrule*[right=\texttt{\texttt{S-Int}}]
{ }
{\Gamma; \Delta \vdash \mathtt{Int} \leq \mathtt{Int} \dashv \Gamma; \Delta \rightsquigarrow \mathtt{Int}}

\inferrule*[right=\texttt{S-Empty}]
{\Gamma; \Delta \vdash_{close}A}
{\Gamma; \Delta \vdash A \leq^+ \square \dashv \Gamma; \Delta \rightsquigarrow \Delta(A)}

\\
\inferrule*[right=\texttt{S-UVar}]
{ }
{\Gamma; \Delta \vdash a \leq a \dashv \Gamma; \Delta \rightsquigarrow a}


\inferrule*[right=\texttt{S-Sol}]
{ }
{\Gamma[\hat{a}=A]; \Delta \vdash a \leq a \dashv \Gamma[\hat{a}=A];  \Delta \rightsquigarrow a}

\inferrule*[right=\texttt{S-Sol-L}]
{\Gamma [\hat{a}=B]; \Delta \vdash B <: A \dashv \Gamma [\hat{a}=B]; \Delta'\rightsquigarrow A'}
{\Gamma [\hat{a}=B]; \Delta \vdash \hat{a} \leq A \dashv \Gamma [\hat{a}=B]; \Delta'\rightsquigarrow A'}

\inferrule*[right=\texttt{S-Sol-R}]
{\Gamma [\hat{a}=B]; \Delta \vdash A <: B \dashv \Gamma [\hat{a}=B]; \Delta'\rightsquigarrow B'}
{\Gamma [\hat{a}=B]; \Delta \vdash A \leq \hat{a} \dashv \Gamma [\hat{a}=B]; \Delta'\rightsquigarrow B'}

\inferrule*[right=\texttt{S-Ex-L}]
{ }
{\Gamma; \Delta[\hat{a}] \vdash \hat{a} \leq^+ A \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow A}

\inferrule*[right=\texttt{S-Ex-R}]
{ }
{\Gamma; \Delta[\hat{a}] \vdash A \leq^- \hat{a} \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow A}

\inferrule*[right=\texttt{S-Ex-Eq-L}]
{\Gamma; \Delta[\hat{a:=A}] \vdash  A \leq^+ B \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow B}
{\Gamma; \Delta[\hat{a:=A}] \vdash \hat{a} \leq^+ B \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow B}


\inferrule*[right=\texttt{S-Ex-Eq-R}]
{\Gamma; \Delta[\hat{a:=A}] \vdash  B \leq^- A \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow B}
{\Gamma; \Delta[\hat{a:=A}] \vdash B \leq^- \hat{a}  \dashv \Gamma; \Delta[\hat{a} = A] \rightsquigarrow B}

\inferrule*[right=\texttt{S-Arr}]
{\Gamma; \Delta \vdash C \leq^* A \dashv \Gamma; \Omega \rightsquigarrow C' \\
 \Gamma; \Omega \vdash B \leq D \dashv \Gamma; \Delta' \rightsquigarrow D'}
{\Gamma; \Delta \vdash A \rightarrow B \leq C \rightarrow D \dashv \Gamma; \Delta' \rightsquigarrow C' \rightarrow D'}
\\

\inferrule*[right=\texttt{S-Term-Close}]
{ \Delta \vdash_{close} A \\
\Gamma \vdash \Delta(A) \Rightarrow e \Rightarrow A'\\
 \Gamma \vdash B \leq^+ \Sigma \dashv \Delta \rightsquigarrow D}
{\Gamma; \Delta \vdash A \rightarrow B \leq^+ \boxed{e} \leadsto \Sigma \dashv \Gamma; \Delta \rightsquigarrow \Delta(A) \rightarrow D}

\inferrule*[right=\texttt{S-Term-Open}]
{ \Delta \vdash_{open} A \\
\Gamma \vdash \square \Rightarrow e \Rightarrow C\\
\Gamma \vdash C \leq^- A \dashv \Omega \rightsquigarrow A' \\
\Omega \vdash B \leq^+ \Sigma \dashv \Delta \rightsquigarrow D}
{\Gamma \vdash A \rightarrow B \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta \rightsquigarrow C \rightarrow D}
\\

\inferrule*[right=\texttt{S-$\forall$}]
{\Gamma; \Delta, a \vdash A \leq B \dashv \Gamma; \Delta', a\rightsquigarrow C}
{\Gamma; \Delta \vdash \forall a.A \leq \forall a.~B \dashv \Gamma; \Delta' \rightsquigarrow \forall a. C}

\inferrule*[right=\texttt{S-$\forall$L}]
{\Gamma; \Delta, \hat{a} \vdash A \leq^+ \boxed{e} \leadsto \Sigma \dashv \Gamma; \Delta', \hat{a}=C \rightsquigarrow B}
{\Gamma; \Delta \vdash \forall a. A \leq^+ \boxed{e} \leadsto \Sigma \dashv \Gamma; \Delta' \rightsquigarrow B}
\end{mathpar}
