\begin{align*}
&\text{Types} \quad\quad &A, B, C, D ::=&~ a \mid \mathtt{Int} \mid A \rightarrow B \mid \forall a.~A\\
&\text{Expressions} \quad \quad &e::=&~ i \mid x \mid \lambda x . ~e \mid e_1~e_2 \mid e : A \mid \Lambda a.~e \mid e ~@ A
\end{align*}

\section{Algo.}

\begin{align*}
&\text{Typing Environments} \quad\quad &\Gamma::=&~ . \mid \Gamma, x : A \mid \Gamma, a\\
&\text{Subtyping Environments} \quad\quad &\Gamma, \Omega, \Delta ::=&~ . \mid \Gamma, x : A \mid \Gamma, a \mid \Gamma , \hat{a} \mid \Gamma, \hat{a} = A\\
    &\text{Contexts} \quad\quad &\Sigma ::=&~  \square  \mid A \mid \boxed{e} \leadsto \Sigma \mid @A \leadsto \Sigma \mid \boxed{A} \leadsto \Sigma\\
    &\text{Generic Consumers} \quad \quad &g::=&~ i \mid x \mid e : A \mid \Lambda a.~e
\end{align*}

\subsection{Typing: $\Gamma \vdash \Sigma \Rightarrow e \Rightarrow A$}

\begin{mathpar}
\inferrule*[right=\texttt{T-Lit}]
{ }
{\Gamma \vdash \square \Rightarrow i \Rightarrow \mathsf{Int}}

\inferrule*[right=\texttt{T-Var}]
{ x : A \in \Gamma}
{\Gamma \vdash \square \Rightarrow x \Rightarrow A}

\inferrule*[right=\texttt{T-Ann}]
{\Gamma \vdash A \Rightarrow e \Rightarrow B}
{\Gamma \vdash \square \Rightarrow e : A \Rightarrow A}

\inferrule*[right=\texttt{T-App}]
{\Gamma \vdash \boxed{e_2} \leadsto \Sigma \Rightarrow e_1 \Rightarrow A \rightarrow B}
{\Gamma \vdash \Sigma \Rightarrow e_1~e_2 \Rightarrow B}

\inferrule*[right=\texttt{T-Lam1}]
{\Gamma, x : A \vdash B \Rightarrow e \Rightarrow C}
{\Gamma \vdash A \rightarrow B \Rightarrow \lambda x.~e \Rightarrow A \rightarrow C}

\inferrule*[right=\texttt{T-Lam2}]
{\Gamma \vdash \square \Rightarrow e_2 \Rightarrow A\\
 \Gamma , x : A \vdash \Sigma \Rightarrow e \Rightarrow B}
{\Gamma \vdash \boxed{e_2} \leadsto \Sigma \Rightarrow \lambda x.~e \Rightarrow A \rightarrow B}

\inferrule*[right=\texttt{T-Lam3}]
{\Gamma , x : A \vdash \Sigma \Rightarrow e \Rightarrow B}
{\Gamma \vdash \boxed{A} \leadsto \Sigma \Rightarrow \lambda x.~e \Rightarrow A \rightarrow B}

\inferrule*[right=\texttt{T-Sub}]
{\Gamma \vdash \square \Rightarrow g \Rightarrow A\\
\Sigma \neq \square\\
 \Gamma; . \vdash A \leq^+ \Sigma \dashv . \rightsquigarrow B}
{\Gamma \vdash \Sigma \Rightarrow g \Rightarrow B}

\inferrule*[right=\texttt{T-TAbs}]
{\Gamma, a \vdash \square \Rightarrow e \Rightarrow A}
{\Gamma \vdash \square \Rightarrow \Lambda a.~e \Rightarrow \forall a.~A}

%\inferrule*[right=\texttt{T-TAbs2}]
%{\Gamma, a \vdash \Sigma  \Rightarrow e \Rightarrow B}
%{\Gamma \vdash \boxed{A} \leadsto \Sigma \Rightarrow \Lambda a. ~e \Rightarrow [A/a]B}

\inferrule*[right=\texttt{T-TApp}]
%{\Gamma \vdash H \Rightarrow e \Rightarrow \forall X.~B}
{\Gamma \vdash @A \leadsto \Sigma \Rightarrow e \Rightarrow \forall a.~B}
{\Gamma \vdash  \Sigma \Rightarrow e ~@A \Rightarrow [A/a]B}
\end{mathpar}

\subsection{Subtyping: $\Gamma; \Delta \vdash A \leq^{\pm} \Sigma \dashv \Delta' \rightsquigarrow A'$}

\begin{mathpar}
\inferrule*[right=\texttt{\texttt{S-Int}}]
{ }
{\Gamma; \Delta \vdash \mathtt{Int} \leq \mathtt{Int} \dashv \Delta  }

\inferrule*[right=\texttt{S-UVar}]
{ }
{\Gamma; \Delta \vdash a \leq a \dashv \Delta }
\\
\inferrule*[right=\texttt{S-Ex-L}]
{ }
{\Gamma; \Delta[\hat{a}] \vdash \hat{a} \leq^+ A \dashv \Delta[\hat{a} = A] }

\inferrule*[right=\texttt{S-Ex-R}]
{ }
{\Gamma; \Delta[\hat{a}] \vdash A \leq^- \hat{a} \dashv \Delta[\hat{a} = A] }

\inferrule*[right=\texttt{S-Ex-Eq-L}]
{ }
{\Gamma; \Delta[\hat{a:=A}] \vdash \hat{a} \leq^+ A \dashv \Delta[\hat{a} = A] }

\inferrule*[right=\texttt{S-Ex-Eq-R}]
{ }
{\Gamma; \Delta[\hat{a:=A}] \vdash A \leq^- \hat{a}  \dashv \Delta[\hat{a} = A] }

\inferrule*[right=\texttt{S-Arr}]
{\Gamma; \Delta \vdash C \leq^* A \dashv \Omega  \\
 \Gamma; \Omega \vdash B \leq D \dashv \Delta' }
{\Gamma; \Delta \vdash A \rightarrow B \leq C \rightarrow D \dashv \Delta' }

\inferrule*[right=\texttt{S-$\forall$}]
{\Gamma; \Delta, a \vdash A \leq B \dashv \Delta', a }
{\Gamma; \Delta \vdash \forall a.A \leq \forall a.~B \dashv \Delta' }

\noindent\makebox[\linewidth]{\rule{16cm}{0.4pt}}

\inferrule*[right=\texttt{S-Empty}]
{\Gamma; \Delta \vdash_{\texttt{close}}A}
{\Gamma; \Delta \vdash A \leq^+ \square \dashv \Delta \rightsquigarrow \Delta(A)}

\inferrule*[right=\texttt{S-Type}]
{\Gamma; \Delta \vdash A \leq B  \dashv \Delta' }
{\Gamma; \Delta \vdash A \leq B  \dashv \Delta' \rightsquigarrow B  }

\inferrule*[right=\texttt{S-Term-Close}]
{ \Delta \vdash_{\texttt{close}} A \\
\Gamma \vdash \Delta(A) \Rightarrow e \Rightarrow A'\\
 \Gamma; \Delta \vdash B \leq^+ \Sigma \dashv \Delta' \rightsquigarrow D}
{\Gamma; \Delta \vdash A \rightarrow B \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta' \rightsquigarrow \Delta(A) \rightarrow D}

\inferrule*[right=\texttt{S-Term-Open1}]
{ \Delta \vdash_{\texttt{open}} A \\
 \texttt{need}(e) \le \texttt{have}(A) \\
\Gamma \vdash *\Gamma(A)* \Rightarrow e \Rightarrow C\\
\Gamma; \Delta \vdash C \leq^- A \dashv \Omega \\
\Gamma; \Omega \vdash B \leq^+ \Sigma \dashv \Delta' \rightsquigarrow D}
{\Gamma; \Delta \vdash A \rightarrow B \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta' \rightsquigarrow C \rightarrow D}

\inferrule*[right=\texttt{S-Term-Open2}]
{ \Delta \vdash_{\texttt{open}} A \\
 \texttt{need}(e) > \texttt{have}(A) \\
\Gamma; \Delta \vdash B \leq^+ \Sigma \dashv \Omega \rightsquigarrow D\\
\Gamma \vdash *\Omega(A)* \Rightarrow e \Rightarrow C\\
\Gamma; \Omega \vdash C \leq^- A \dashv \Delta'}
{\Gamma; \Delta \vdash A \rightarrow B \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta' \rightsquigarrow C \rightarrow D}

\inferrule*[right=\texttt{S-$\forall$L}]
{\Gamma; \Delta, \hat{a} \vdash A \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta', \hat{a}=C \rightsquigarrow B}
{\Gamma; \Delta \vdash \forall a. A \leq^+ \boxed{e} \leadsto \Sigma \dashv \Delta' \rightsquigarrow B}

\inferrule*[right=\texttt{S-TApp}]
{ \Gamma; \Delta, a =B \vdash A \leq^+ \Sigma \dashv \Delta, a = B \rightsquigarrow C}
{\Gamma; \Delta \vdash \forall a. A \leq^+ @B \leadsto \Sigma \dashv \Delta \rightsquigarrow \forall a. C}

\end{mathpar}
